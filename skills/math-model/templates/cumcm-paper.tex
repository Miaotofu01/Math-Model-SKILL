%% CUMCM 论文模板（2026 修订版，math-model skill 内置）
%% 由 2026-08 窃电识别赛题实战经验提炼：
%%   - 摘要用 \section*{摘 要}（章节式标题，无编号）——不用 abstract 环境，
%%     避免环境自动标题与章节标题重复出现"双摘要"
%%   - 章节标题居中（titlesec，article+ctex 可用；ctexart 用 \ctexset）
%%   - 关键答案数值用 $\bm{...}$ 加粗（摘要/正文统一）
%%   - 代码附录用 pythonstyle（彩色高亮），\lstinputlisting 直接引用文件
%%   - 支撑材料清单为真实交付物模板（非占位式）
%%   - 身份信息禁令：全文不得出现学校/姓名/绝对路径（路径用 ~ 展开）
%%
%% 占位符（组装脚本用 replace 填入；注意：下列占位符说明不含 @ 字符，
%% 只有正文区的占位符用 @XXX@ 包裹，避免替换污染注释）：
%%   TITLE        竞赛题名（如"全国大学生数学建模竞赛B题"）
%%   PAPER_TITLE  论文题目
%%   SUBTITLE     副标题（方法名）
%%   ABSTRACT     摘要内容（以 \section*{摘 要} 开头，含关键词行）
%%   BODY         正文章节（问题重述/假设/模型/分析/结论）
%%   REFERENCES   thebibliography 内容
%%   MATERIALS    支撑材料清单 \item 列表
%%   SUSPECT_LISTS 疑似名单表（如有；无则整节删除）
%%   CODE_ENTRIES 代码 \subsection + \lstinputlisting 条目
%%   CODE_DIR     code/ 目录相对路径
%% ============================================================
\documentclass[12pt, a4paper]{article}

%% ======================== 宏包导入 ========================
\usepackage{ctex}                    % 中文支持
\usepackage[margin=2.5cm]{geometry}  % 页边距（CUMCM 规范：≥2.5cm）
\usepackage{amsmath, amssymb, amsthm}% 数学公式
\usepackage{bm}                      % 粗体数学（摘要关键数值 $\bm{...}$）
\usepackage{graphicx}                % 插图
\usepackage{subcaption}              % 子图
\usepackage{booktabs}                % 三线表
\usepackage{longtable}               % 长表格
\usepackage{array}                   % 表格增强
\usepackage{multirow}                % 跨行单元格
\usepackage{float}                   % 浮动体控制 [H]
\usepackage{hyperref}                % 超链接
\hypersetup{colorlinks=true,linkcolor=black,citecolor=black,urlcolor=black}
\usepackage{enumerate}               % 列表
\usepackage{enumitem}                % 列表定制
\usepackage{listings}                % 代码高亮
\usepackage{xcolor}                  % 颜色
\usepackage{caption}                 % 标题定制
\usepackage{fancyhdr}                % 页眉页脚
\usepackage{setspace}                % 行距
\usepackage{cite}                    % 参考文献引用

%% ======================== 章节标题居中 ========================
%% article+ctex 组合下 ctex 不接管 \section 格式，必须用 titlesec；
%% 若改用 ctexart 文档类，则用 \ctexset{section={format={\centering}}} 替代本块。
\usepackage{titlesec}
\titleformat{\section}{\centering\Large\bfseries}{\thesection}{0.6em}{}
\titleformat{name=\section,numberless}{\centering\Large\bfseries}{}{0em}{}

%% ======================== 代码样式（彩色高亮） ========================
\definecolor{codegray}{rgb}{0.5,0.5,0.5}
\definecolor{codepurple}{rgb}{0.58,0,0.82}
\definecolor{codeblue}{rgb}{0.13,0.29,0.53}
\definecolor{backcolour}{rgb}{0.97,0.97,0.97}

\lstdefinestyle{pythonstyle}{
    backgroundcolor=\color{backcolour},
    commentstyle=\color{codegray}\itshape,
    keywordstyle=\color{codeblue}\bfseries,
    stringstyle=\color{codepurple},
    basicstyle=\ttfamily\footnotesize,
    breakatwhitespace=false,
    breaklines=true,
    captionpos=b,
    keepspaces=true,
    numbers=left,
    numbersep=5pt,
    numberstyle=\tiny\color{codegray},
    showspaces=false,
    showstringspaces=false,
    showtabs=false,
    tabsize=4,
    frame=single,
    rulecolor=\color{black!20},
    language=Python
}

%% ======================== 页面样式（CUMCM 合规） ========================
%% 摘要专用页起页码，页脚中部阿拉伯数字；正文无目录；页眉简洁不含身份信息
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0pt}

\captionsetup{font=small, labelfont=bf}
\setstretch{1.25}

%% ======================== 自定义命令 ========================
\newtheorem{assumption}{假设}[section]
\newtheorem{definition}{定义}[section]
\newtheorem{theorem}{定理}[section]

\newcommand{\prob}[1]{\mathbf{P}\left(#1\right)}
\newcommand{\E}[1]{\mathbb{E}\left[#1\right]}
\newcommand{\norm}[1]{\left\|#1\right\|}

\begin{document}

%% ======================== 摘要专用页（第一页，页码从 1 开始） ========================

\begin{center}
{\large \textbf{@TITLE@}}
\end{center}
\vspace{0.5cm}
\begin{center}
{\Large\textbf{@PAPER_TITLE@}}\\[0.3em]
{\large @SUBTITLE@}
\end{center}

%% 摘要：\section*（无编号，与其他章节同款实现，避免 abstract 环境的"双摘要"）
@ABSTRACT@

\newpage

%% ======================== 正文（≤20页，无目录） ========================
@BODY@

%% ======================== 参考文献（结论之后、附录之前） ========================
%% 注意：thebibliography 环境自带"参考文献"标题，此处不要再加 \section*{参考文献}，
%% 否则出现"参考文献"双标题。
@REFERENCES@

\newpage
%% ======================== 附录（\appendix 切换为 A/B/C 字母编号，不与正文连续计数） ========================
\appendix
%% 附录编号与正文独立：表/图/公式按 \Alph{section}.N 编号（A.1、B.2...），计数器重置从 1 起
\setcounter{table}{0}
\setcounter{figure}{0}
\setcounter{equation}{0}
\renewcommand{\thetable}{\Alph{section}.\arabic{table}}
\renewcommand{\thefigure}{\Alph{section}.\arabic{figure}}
\renewcommand{\theequation}{\Alph{section}.\arabic{equation}}
